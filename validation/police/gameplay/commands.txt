Canonical clean rebuild of the tested final source:
  mkdir -p build/police-validation/reproduce-{strict,sanitized}
  for mode in strict sanitized; do
    flags=(-std=c++20 -Wall -Wextra -Wpedantic -Werror)
    if [ "$mode" = sanitized ]; then
      flags+=(-O1 -g -fno-omit-frame-pointer -fsanitize=address,undefined)
    else
      flags+=(-O2)
    fi
    output="build/police-validation/reproduce-$mode"
    for source in game game_law law law_navigation pedestrians visuals world world_geometry; do
      g++ "${flags[@]}" -c "src/$source.cpp" -o "$output/$source.o" || exit 1
    done
    for target in game_police game pedestrian law law_navigation; do
      g++ "${flags[@]}" "tests/${target}_tests.cpp" "$output/"*.o -o "$output/${target}_tests" || exit 1
      ASAN_OPTIONS=detect_leaks=1 UBSAN_OPTIONS=halt_on_error=1 "$output/${target}_tests" > "$output/$target.log" 2>&1 || exit 1
    done
  done
